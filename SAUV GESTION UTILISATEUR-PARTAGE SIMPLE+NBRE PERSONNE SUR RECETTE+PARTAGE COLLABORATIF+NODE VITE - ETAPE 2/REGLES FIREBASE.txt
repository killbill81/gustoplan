  rules_version = '2';
  service cloud.firestore {
    match /databases/{database}/documents {

      // --- Fonctions d'aide ---
      function isAuth() {
        return request.auth != null;
      }

      // --- Collections ---

      // Utilisateurs
      match /users/{userId} {
        allow read: if isAuth();
        allow write: if request.auth.uid == userId;
      }

      // Recettes & Ingrédients (publics pour les utilisateurs connectés)
      match /recipes/{recipeId} {
        allow read, write: if isAuth();
      }
      match /ingredients/{ingredientId} {
        allow read, write: if isAuth();
      }
      match /ingredient_categories/{categoryId} {
        allow read, write: if isAuth();
      }

    // Plans (personnels et collaboratifs)
          match /plans/{planId} {
        // Un utilisateur peut lire un plan s'il en est le propriétaire (userId)
        // ou s'il fait partie des collaborateurs.
        allow read: if request.auth.uid == resource.data.userId
                    || (resource != null && request.auth.uid in resource.data.collaborators);

        // Un utilisateur peut créer un plan s'il s'en déclare propriétaire.
        allow create: if request.auth.uid == request.resource.data.userId;

        // Le propriétaire ou un collaborateur peut mettre à jour le plan.
        allow update: if request.auth.uid == resource.data.userId
                      || request.auth.uid in resource.data.collaborators;

        // Seul le propriétaire peut supprimer un plan.
        allow delete: if request.auth.uid == resource.data.userId;
      }
    
// Listes de courses (personnelles)
  match /shopping_lists/{listId} {
    allow read: if request.auth.uid == resource.data.userId;

    // Pour créer/modifier, on vérifie les nouvelles données
    allow create, update: if request.auth.uid == request.resource.data.userId;

    // Pour supprimer, on vérifie les données existantes
    allow delete: if request.auth.uid == resource.data.userId;
  }

      // Liste de courses active
      match /active_shopping_list/{userId} {
          allow read, write: if request.auth.uid == userId;
      }

      // Invitations (partages & demandes d'amis)
      match /shares/{shareId} {
        allow create: if request.auth.uid == request.resource.data.senderId;
        allow read, update, delete: if isAuth() && (request.auth.uid == resource.data.senderId || request.auth.uid ==
  resource.data.receiverId);
      }
      match /friend_requests/{reqId} {
         allow create: if request.auth.uid == request.resource.data.senderId;
         allow read, update, delete: if isAuth() && (request.auth.uid == resource.data.senderId || request.auth.uid ==
  resource.data.receiverId);
      }

      // --- Futur : Partage collaboratif ---
      match /collaborative_plans/{collabPlanId} {
        function isMember() {
            return request.auth.uid in resource.data.members;
        }
        allow read, write: if isAuth() && isMember();
        allow create: if isAuth() && request.auth.uid == request.resource.data.ownerId && request.auth.uid in request.resource.data.members;
      }
    }
  }
